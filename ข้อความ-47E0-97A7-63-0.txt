        function renderProducts() {
            const grid = document.getElementById('productsGrid');
            const featuredGrid = document.getElementById('featuredProductsGrid');
            const searchKeyword = document.getElementById('searchInput') ? document.getElementById('searchInput').value.toLowerCase() : '';

            let filtered = products.filter(p => {
                const matchCat = selectedCategory === 'all' || p.category === selectedCategory;
                const matchSearch = p.name.toLowerCase().includes(searchKeyword) || p.description.toLowerCase().includes(searchKeyword);
                return matchCat && matchSearch;
            });

            // Render Catalog Grid
            if (grid) {
                grid.innerHTML = '';
                if (filtered.length === 0) {
                    grid.innerHTML = `<div class="col-span-full text-center py-12 text-slate-400">ไม่พบสินค้าที่คุณค้นหา 🌊</div>`;
                } else {
                    filtered.forEach(prod => {
                        grid.appendChild(createProductCardHTML(prod));
                    });
                }
            }

            // Render Featured Showcase on Home
            if (featuredGrid) {
                featuredGrid.innerHTML = '';
                products.slice(0, 3).forEach(prod => {
                    featuredGrid.appendChild(createProductCardHTML(prod));
                });
            }
        }

                        <button onclick="deleteProduct('${p.id}')" class="p-2 rounded-lg bg-red-500/20 hover:bg-red-500/40 text-red-300 text-xs">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function renderAdminUsersTable() {
            const tbody = document.getElementById('adminUsersTable');
            if (!tbody) return;

            tbody.innerHTML = '';
            sampleUsers.forEach(u => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-cyan-900/20 transition';
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${u.name}</td>
                    <td class="p-3 text-slate-300 text-xs">${u.email}</td>
                    <td class="p-3">
                        <span class="px-2.5 py-1 rounded-full text-xs font-semibold ${u.role === 'reseller' ? 'bg-cyan-500/30 text-cyan-200 border border-cyan-400/30' : 'bg-slate-700 text-slate-300'}">
                            ${u.role === 'reseller' ? '🌊 ตัวแทนจำหน่าย' : 'ผู้ซื้อทั่วไป'}
                        </span>
                    </td>
                    <td class="p-3 font-bold text-cyan-300">฿${u.totalSpent}</td>
                    <td class="p-3 text-center">
                        <button onclick="toggleUserRole('${u.id}')" class="px-3 py-1 rounded-full bg-cyan-500/20 hover:bg-cyan-500/40 text-cyan-300 text-xs border border-cyan-400/30">
                            สลับยศ 🌊
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function toggleUserRole(userId) {
            const user = sampleUsers.find(u => u.id === userId);
            if (user) {
                user.role = user.role === 'reseller' ? 'customer' : 'reseller';
                renderAdminUsersTable();
                showToast('ปรับยศเรียบร้อย 🌊', `เปลี่ยนยศให้ ${user.name} สำเร็จ`);
            }
        }

        function openAddProductModal() {
            document.getElementById('addProductModal').classList.remove('hidden');
        }

        function saveNewProduct() {
            const name = document.getElementById('newProdName').value;
            const category = document.getElementById('newProdCategory').value;
            const icon = document.getElementById('newProdIcon').value || 'https://cdn-icons-png.flaticon.com/512/888/888857.png';
            const priceNormal = parseFloat(document.getElementById('newProdPriceNormal').value) || 99;
            const priceReseller = parseFloat(document.getElementById('newProdPriceReseller').value) || 69;
            const desc = document.getElementById('newProdDesc').value || 'แอปพรีเมียมคุณภาพสูง รับประกันตลอดยุคใช้งาน';

            if (!name) {
                showToast('กรอกชื่อแอป!', 'โปรดระบุชื่อสินค้า', 'fa-triangle-exclamation');
                return;
            }

            const newProd = {
                id: 'prod_' + (products.length + 1),
                name: name,
                category: category,
                icon: icon,
                description: desc,
                packages: [
                    { name: '1 เดือน (แพ็กเกจพรีเมียม)', priceNormal: priceNormal, priceReseller: priceReseller, stock: 10 }
                ]
            };

            products.push(newProd);
            closeModal('addProductModal');
            renderProducts();
            renderAdminProductsTable();

            showToast('เพิ่มสินค้าใหม่แล้ว! 🌊', `เพิ่ม ${name} เข้าสู่ระบบเรียบร้อย`);
        }

        function deleteProduct(prodId) {
            products = products.filter(p => p.id !== prodId);
            renderProducts();
            renderAdminProductsTable();
            showToast('ลบสินค้าสำเร็จ 🌊', 'ลบสินค้าออกจากระบบแล้ว');
        }

        function savePaymentSettings() {
            const no = document.getElementById('adminPromptPayNo').value;
            const name = document.getElementById('adminPromptPayName').value;
            document.getElementById('promptPayNameDisplay').innerText = name;
            showToast('บันทึกเรียบร้อย 🌊', 'อัปเดตข้อมูลการรับเงิน PromptPay แล้ว');
        }

        function showToast(title, message, iconClass = 'fa-water') {

            document.getElementById('toastTitle').innerText = title;

            toast.classList.remove('translate-x-72', 'opacity-0', 'pointer-events-none');
            
            setTimeout(() => {
                toast.classList.add('translate-x-72', 'opacity-0', 'pointer-events-none');
